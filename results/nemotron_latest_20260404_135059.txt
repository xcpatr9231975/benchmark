def sum_of_integers(integer_list):
    """
    Calculates the sum of all integers in a given list.
    
    Args:
        integer_list (list): A list containing integers.
    
    Returns:
        int: The sum of all integers in the list. If the list is empty, returns 0.
    
    Raises:
        TypeError: If the input is not a list or if any element in the list is not an integer.
    """
    # Check if input is a list
    if not isinstance(integer_list, list):
        raise TypeError("Input must be a list.")
    
    # Initialize sum variable to 0
    total_sum = 0
    
    # Iterate over each element in the list
    for element in integer_list:
        # Check if element is an integer
        if not isinstance(element, int):
            raise TypeError("All elements in the list must be integers.")
        
        # Add the integer to the sum
        total_sum += element
    
    return total_sum
